// Google Apps Script for Google Sheets Integration
// To use this script:
// 1. Create a new Google Sheet
// 2. Go to Extensions > Apps Script
// 3. Copy this code into the script editor
// 4. Set up the headers in your sheet: Name, Email, Phone, Guests, Date, Time, Occasion, Special Requests, Timestamp
// 5. Deploy as Web App (Publish > Deploy as Web App)
// 6. Copy the Web App URL and paste it in the script.js file

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var timestamp = new Date();

    // Customer data has no password: Firebase Authentication stores passwords securely.
    if (data.type === 'customer') {
      getOrCreateSheet_('Customers', ['Customer ID', 'Name', 'Email', 'Marketing Opt-in', 'Timestamp']).appendRow([
        data.customerId, data.name, data.email, data.marketingOptIn, timestamp
      ]);
    } else if (data.type === 'order') {
      getOrCreateSheet_('Customer Orders', ['Customer ID', 'Email', 'Item', 'Price', 'Status', 'Timestamp']).appendRow([
        data.customerId, data.email, data.itemName, data.price, data.status, timestamp
      ]);
    } else {
      getOrCreateSheet_('Reservations', ['Name', 'Email', 'Phone', 'Guests', 'Date', 'Time', 'Occasion', 'Special Requests', 'Timestamp']).appendRow([
        data.name, data.email, data.phone, data.guests, data.date, data.time, data.occasions, data.requests, timestamp
      ]);
    }
    
    return ContentService
      .createTextOutput(JSON.stringify({ 'result': 'success', 'message': 'Reservation submitted successfully!' }))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (error) {
    return ContentService
      .createTextOutput(JSON.stringify({ 'result': 'error', 'message': error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function getOrCreateSheet_(name, headers) {
  var spreadsheet = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = spreadsheet.getSheetByName(name);
  if (!sheet) {
    sheet = spreadsheet.insertSheet(name);
    sheet.appendRow(headers);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold');
    sheet.setFrozenRows(1);
  }
  return sheet;
}

// Optional: Function to set up the sheet with headers
function setupSheet() {
  getOrCreateSheet_('Reservations', ['Name', 'Email', 'Phone', 'Guests', 'Date', 'Time', 'Occasion', 'Special Requests', 'Timestamp']);
  getOrCreateSheet_('Customers', ['Customer ID', 'Name', 'Email', 'Marketing Opt-in', 'Timestamp']);
  getOrCreateSheet_('Customer Orders', ['Customer ID', 'Email', 'Item', 'Price', 'Status', 'Timestamp']);
}
